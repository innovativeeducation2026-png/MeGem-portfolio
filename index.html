1. Data Cleaning Script (CSV / Spreadsheet Prep)

Normalizes whitespace, strips currency symbols, drops blank rows, and outputs clean tabular records. 

import csv, io

raw_csv = """item,price,qty
  Widget A , "$12.50", 3
  Widget B , "£8.00",
  Widget C , "$4.25", 10
"""

reader = csv.DictReader(io.StringIO(raw_csv))
cleaned = []
for row in reader:
    name = (row["item"] or "").strip()
    price_str = (row["price"] or "").replace("$", "").replace("£", "").strip()
    qty_str = (row["qty"] or "").strip()
    if name and price_str and qty_str:
        price = float(price_str)
        qty = int(qty_str)
        cleaned.append({
            "item": name,
            "unit_price": f"{price:.2f}",
            "qty": qty,
            "total": f"{price * qty:.2f}"
        })

for entry in cleaned:
    print(entry)

# Cleaned output:
# {'item': 'Widget A', 'unit_price': '12.50', 'qty': 3, 'total': '37.50'}
# {'item': 'Widget C', 'unit_price': '4.25', 'qty': 10, 'total': '42.50'}
